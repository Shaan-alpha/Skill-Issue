# v0.4.0 — AI Narrative Layer Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship the v0.4.0 slice — Roast Mode and Mentor Mode produce streaming, on-voice narrative wrapping every Report without altering scores. SSE endpoint on the backend, mode-toggle pill row + streaming text card on the frontend, in-process LRU cache, per-day OpenAI budget with deterministic fallback when exhausted.

**Spec:** [`docs/superpowers/specs/2026-05-16-v0.4.0-narrative-design.md`](../specs/2026-05-16-v0.4.0-narrative-design.md). When this plan and the spec disagree, the spec wins — fix the plan.

**Architecture:**
- Backend `app/narrative/` holds six small modules (`service`, `prompts`, `llm`, `cache`, `budget`, `fallback`). The single boundary to OpenAI lives in `llm.py`; tests inject a `FakeNarrativeLLM` and never hit the network. `service.py` is an async generator the `/narrative/{username}` SSE route awaits. Cache is a 256-entry LRU keyed by `(username, scores_hash, mode)`. Budget is a per-day counter with UTC-midnight rollover; when exceeded a deterministic `fallback.py` template streams in place of the live narrative.
- Frontend replaces the right-hand "Engineering Report" hero card from v0.3.0 with a `NarrativeCard` that owns the SSE lifecycle. Pill tabs at the top, streaming text below, animated typing cursor while tokens arrive, fallback toast when the AI is offline.

**Tech Stack:** Python 3.12 · FastAPI (`StreamingResponse`) · Pydantic v2 · `openai>=1.50` (async streaming) · pytest + respx · ruff · Next.js 16 (Turbopack) · React 19 · TypeScript · Tailwind v4 · framer-motion (`LazyMotion`/`m.*`, `layoutId` for the toggle indicator) · Base UI · lucide-react.

```python
import hashlib
import json
from collections import OrderedDict

from app.models import Report


class NarrativeCache:
    """In-process LRU mapping (username, scores_hash, mode) -> narrative text.

    Keys are the *complete* narrative the LLM produced. Re-streams synthesise
    SSE events from the cached string in `service.py`.
    """

    def __init__(self, max_entries: int = 256) -> None:
        self._store: OrderedDict[str, str] = OrderedDict()
        self._max = max_entries

    @staticmethod
    def key(username: str, scores_hash: str, mode: str) -> str:
        return f"{username}:{scores_hash}:{mode}"

    @staticmethod
    def scores_hash(report: Report) -> str:
        """Stable 16-char hex digest over the parts of the report the LLM sees.

        Equivalent reports (same total, tier, badges, per-bucket points) hash
        to the same value regardless of badge insertion order.
        """
        payload = json.dumps(
            {
                "total": report.total,
                "tier": report.tier.name,
                "badges": sorted(b.slug for b in report.badges),
                "breakdown": {
                    "repo_quality": report.breakdown.repo_quality.points,
                    "engineering_maturity": report.breakdown.engineering_maturity.points,
                    "oss_collab": report.breakdown.oss_collab.points,
                    "consistency": report.breakdown.consistency.points,
                    "recruiter_signal": report.breakdown.recruiter_signal.points,
                    "learning_trajectory": report.breakdown.learning_trajectory.points,
                },
            },
            sort_keys=True,
        )
        return hashlib.sha256(payload.encode()).hexdigest()[:16]

    def get(self, key: str) -> str | None:
        if key not in self._store:
            return None
        self._store.move_to_end(key)
        return self._store[key]

    def put(self, key: str, value: str) -> None:
        if key in self._store:
            self._store.move_to_end(key)
        self._store[key] = value
        if len(self._store) > self._max:
            self._store.popitem(last=False)
```

```python
from datetime import UTC, datetime, timedelta
from threading import Lock


def _now() -> datetime:
    """Patched in tests; one source of truth for the wall-clock."""
    return datetime.now(UTC)


def _today_iso(now: datetime) -> str:
    return now.strftime("%Y-%m-%d")


def _next_midnight_iso(now: datetime) -> str:
    tomorrow = (now + timedelta(days=1)).date()
    return f"{tomorrow.isoformat()}T00:00:00Z"


class DailyBudget:
    """In-process per-day request counter, UTC midnight rollover.

    Multi-instance: per-Vercel-Function-instance, intentionally. The true
    global cap is `limit * instance_count`. Redis-backed shared counter
    lands with v0.8.0 caching.
    """

    def __init__(self, limit: int) -> None:
        self._limit = limit
        self._lock = Lock()
        now = _now()
        self._day = _today_iso(now)
        self._count = 0

    def try_consume(self) -> tuple[bool, int, str]:
        """Return (allowed, remaining_after, resets_at_iso)."""
        with self._lock:
            now = _now()
            today = _today_iso(now)
            if today != self._day:
                self._day = today
                self._count = 0
            if self._count >= self._limit:
                return (False, 0, _next_midnight_iso(now))
            self._count += 1
            return (True, self._limit - self._count, _next_midnight_iso(now))
```

from typing import Literal

from app.models import Report

Mode = Literal["roast", "mentor"]


_ROAST = (
    "The AI narrator clocks out at midnight UTC. Your score and tier are still "
    "100% real — {tier_name} at {sub_rank}/100 in tier, {total} of 100 across six "
    "signals, {badge_count} badge(s) earned. Come back tomorrow for the proper "
    "roast — the deterministic engine isn't going anywhere."
)

_MENTOR = (
    "The AI mentor is off the clock until midnight UTC. The score still has plenty "
    "to say: you're at {total} of 100, landing in {tier_name} ({sub_rank}/100 in "
    "tier). {badge_count} badge(s) recognise specific strengths — focus on the "
    "lowest-scoring bucket between now and the next analysis."
)


def fallback_narrative(mode: Mode, report: Report) -> str:

from collections.abc import AsyncIterator

from openai import AsyncOpenAI


class NarrativeLLM:
    """Single-file boundary to OpenAI. Swap providers here without touching
    the rest of the codebase."""

    def __init__(self, *, api_key: str, model: str) -> None:
        self._client = AsyncOpenAI(api_key=api_key)
        self._model = model

    async def stream_chat(
        self,
        messages: list[dict[str, str]],
        *,
        max_output_tokens: int = 320,
    ) -> AsyncIterator[str]:
        stream = await self._client.chat.completions.create(
            model=self._model,
            messages=messages,
            stream=True,
            max_tokens=max_output_tokens,
            temperature=0.7,
        )
        async for chunk in stream:
            delta = chunk.choices[0].delta.content
            if delta:
                yield delta


class FakeNarrativeLLM:
    """Test double. Yields a scripted list of tokens, records calls."""

    def __init__(self, tokens: list[str]) -> None:
        self._tokens = tokens
        self.calls = 0
        self.last_messages: list[dict[str, str]] | None = None

    async def stream_chat(
        self,
        messages: list[dict[str, str]],
        *,
        max_output_tokens: int = 320,
    ) -> AsyncIterator[str]:
        self.calls += 1
        self.last_messages = messages
        for tok in self._tokens:
            yield tok
```

from collections.abc import AsyncIterator
from dataclasses import dataclass
from typing import Any, Literal, Protocol

from app.models import Report
from app.narrative.budget import DailyBudget
from app.narrative.cache import NarrativeCache
from app.narrative.fallback import fallback_narrative
from app.narrative.prompts import Mode, build_messages

# Chunk size used when re-streaming cached or fallback text. Picked so the
# typing animation paces similarly to a live LLM stream.
_REPLAY_CHUNK = 32


class _LLM(Protocol):
    async def stream_chat(
        self,
        messages: list[dict[str, str]],
        *,
        max_output_tokens: int = 320,
    ) -> AsyncIterator[str]: ...


@dataclass
class StreamEvent:
    """SSE-shaped event the route translates to wire format."""

    kind: Literal["token", "fallback", "done"]
    data: Any  # str for token, dict for fallback/done


async def stream_narrative(
    *,
    report: Report,
    mode: Mode,
    llm: _LLM,
    cache: NarrativeCache,
    budget: DailyBudget,
) -> AsyncIterator[StreamEvent]:
    key = NarrativeCache.key(report.username, NarrativeCache.scores_hash(report), mode)

    cached = cache.get(key)
    if cached is not None:
        async for ev in _replay(cached, kind="token"):
            yield ev
        yield StreamEvent("done", {"mode": mode, "cached": True, "tokens": len(cached)})
        return

    allowed, remaining, resets_at = budget.try_consume()
    if not allowed:
        text = fallback_narrative(mode, report)
        yield StreamEvent(
            "fallback",
            {"reason": "daily_limit", "resets_at": resets_at, "remaining": remaining},
        )
        async for ev in _replay(text, kind="token"):
            yield ev
        yield StreamEvent("done", {"mode": mode, "cached": False, "tokens": 0, "fallback": True})
        return

    messages = build_messages(mode, report)
    buf: list[str] = []
    try:
        async for token in llm.stream_chat(messages):
            buf.append(token)
            yield StreamEvent("token", token)
    except Exception:
        # Upstream LLM failure: serve fallback so the UI never crashes
        text = fallback_narrative(mode, report)
        yield StreamEvent("fallback", {"reason": "upstream_error"})
        async for ev in _replay(text, kind="token"):
            yield ev
        yield StreamEvent("done", {"mode": mode, "cached": False, "tokens": 0, "fallback": True})
        return

    full = "".join(buf)
    cache.put(key, full)
    yield StreamEvent("done", {"mode": mode, "cached": False, "tokens": len(full)})


async def _replay(text: str, *, kind: Literal["token"]) -> AsyncIterator[StreamEvent]:
    """Yield text in fixed-size chunks so cached/fallback re-streams pace
    similarly to a live LLM stream."""
    for i in range(0, len(text), _REPLAY_CHUNK):
        yield StreamEvent(kind, text[i : i + _REPLAY_CHUNK])
```

Create `frontend/src/lib/sse.ts`:

```ts
export type SSEEvent = "token" | "fallback" | "done" | "error";

export interface SSEHandlers {
  onToken?: (text: string) => void;
  onFallback?: (info: unknown) => void;
  onDone?: (info: unknown) => void;
  onError?: (err: unknown) => void;
}

/**
 * Subscribe to a server-sent event stream. Returns a cleanup function.
 *
 * Native EventSource only supports GET and no custom headers. The backend
 * exposes /narrative as GET with no auth so this is sufficient.
 */
export function subscribeSSE(url: string, handlers: SSEHandlers): () => void {
  const es = new EventSource(url);

  es.addEventListener("token", (e) => {
    handlers.onToken?.((e as MessageEvent).data);
  });
  es.addEventListener("fallback", (e) => {
    try {
      handlers.onFallback?.(JSON.parse((e as MessageEvent).data));
    } catch {
      handlers.onFallback?.((e as MessageEvent).data);
    }
  });
  es.addEventListener("done", (e) => {
    try {
      handlers.onDone?.(JSON.parse((e as MessageEvent).data));
    } catch {
      handlers.onDone?.((e as MessageEvent).data);
    }
    es.close();
  });
  es.addEventListener("error", (e) => {
    handlers.onError?.(e);
    es.close();
  });

  return () => es.close();
}
```
